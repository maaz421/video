1. Linear Search with Complexity Estimation

Problem Statement: Perform a Linear Search on an array of integers and estimate its step count ($t$) and memory consumption ($s$).

Corrected C++ Code

#include <iostream>
using namespace std;

int main() {
    int n, i, key, s, t = 0;
    cout << "Enter the number of elements: ";
    cin >> n;

    int a[n];
    cout << "Enter elements:\n";
    t = t + 2;
    for (i = 0; i < n; i++) {
        t = t + 2;
        cin >> a[i];
    }

    cout << "Enter element to search: ";
    cin >> key;

    s = sizeof(n) + sizeof(i) + sizeof(key) + sizeof(s) + sizeof(t) + sizeof(a);
    cout << "Space complexity is " << s << " bytes\n";

    t = t + 2;
    for (i = 0; i < n; i++) {
        t = t + 3;
        if (a[i] == key) {
            cout << "Element found at position " << i + 1 << "\n";
            cout << "Time complexity count (t) is " << t << "\n";
            return 0;
        }
    }
    cout << "Element not found\n";
    cout << "Time complexity count (t) is " << t << "\n";
    return 0;
}

2. Fibonacci Series with Complexity Estimation

Problem Statement: Generate the Fibonacci series up to $n$ terms and calculate its complexity parameters.

Corrected C++ Code

#include <iostream>
using namespace std;

int main() {
    int n, i, a = 0, b = 1, c, t = 0, s;
    cout << "Enter number of terms: ";
    cin >> n;

    s = sizeof(n) + sizeof(i) + sizeof(a) + sizeof(b) + sizeof(c) + sizeof(t) + sizeof(s);

    for (i = 1; i <= n; i++) {
        t = t + 2;
        cout << a << "\n";
        c = a + b;
        a = b;
        b = c;
        t = t + 4;
    }

    cout << "Time complexity count (t) is " << t << "\n";
    cout << "Space complexity is " << s << " bytes\n";
    return 0;
}
